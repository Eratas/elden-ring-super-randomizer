# Windows PE version resource for EldenRingSuperRandomizer.exe, passed to PyInstaller with --version-file.
#
# Why this exists: an unsigned PyInstaller executable with NO version resource looks generic, and
# generic unsigned binaries are exactly what antivirus heuristics and Microsoft SmartScreen flag most
# aggressively.  Embedding who built the tool, what it is and its version gives the file a stable
# identity that reputation systems and AV engines can key on, which measurably reduces false
# positives.  Signing the exe (Authenticode) is the stronger measure - see README "Antivirus notes".
#
# Keep the version tuple in step with tools/version.py (__version__).  PyInstaller execs this file in
# a namespace that provides VSVersionInfo / FixedFileInfo / StringFileInfo / StringTable /
# StringStruct / VarFileInfo / VarStruct, so it must not import them itself.

    flags=0x0,

    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
